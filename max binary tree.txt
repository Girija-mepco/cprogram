// Helper function to allocate a new node
struct TreeNode* newNode(int val) {
    struct TreeNode* node = (struct TreeNode*)malloc(sizeof(struct TreeNode));
    node->val = val;
    node->left = NULL;
    node->right = NULL;
    return node;
}

// Recursive helper function to build the tree
struct TreeNode* buildTree(int* nums, int left, int right) {
    if (left > right) {
        return NULL;
    }
    
    // Find the index of the maximum value
    int maxIdx = left;
    for (int i = left + 1; i <= right; i++) {
        if (nums[i] > nums[maxIdx]) {
            maxIdx = i;
        }
    }
    
    // Create the root node with the maximum value
    struct TreeNode* root = newNode(nums[maxIdx]);
    
    // Recursively build subtrees
    root->left = buildTree(nums, left, maxIdx - 1);
    root->right = buildTree(nums, maxIdx + 1, right);
    
    return root;
}

// Main function required by the judge
struct TreeNode* constructMaximumBinaryTree(int* nums, int numsSize) {
    return buildTree(nums, 0, numsSize - 1);
}
