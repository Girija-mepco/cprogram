#include <stdio.h>
#include <stdlib.h>
#include <string.h>
int compare(const void* a, const void* b) {
    char s1[24];
    char s2[24];
    sprintf(s1, "%s%s", *(const char**)a, *(const char**)b);
    sprintf(s2, "%s%s", *(const char**)b, *(const char**)a);
    return strcmp(s2, s1); 
}
char* largestNumber(int* nums, int numsSize) {
    char** numStrs = (char**)malloc(numsSize * sizeof(char*));
    int totalLength = 0;
    for (int i = 0; i < numsSize; i++) {
        numStrs[i] = (char*)malloc(12 * sizeof(char)); 
        sprintf(numStrs[i], "%d", nums[i]);
        totalLength += strlen(numStrs[i]);
    }
    qsort(numStrs, numsSize, sizeof(char*), compare);
    if (strcmp(numStrs[0], "0") == 0) {
        char* result = (char*)malloc(2 * sizeof(char));
        strcpy(result, "0");
        for (int i = 0; i < numsSize; i++) free(numStrs[i]);
        free(numStrs);   
        return result;
    }
    char* result = (char*)malloc((totalLength + 1) * sizeof(char));
    result[0] = '\0'; 
    for (int i = 0; i < numsSize; i++) {
        strcat(result, numStrs[i]);
        free(numStrs[i]); 
    }   
    free(numStrs);
    return result;
}
