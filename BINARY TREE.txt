/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     struct TreeNode *left;
 *     struct TreeNode *right;
 * };
 */

/**
 * Helper function to calculate the max height (depth - 1) of the tree.
 */
int getHeight(struct TreeNode* root) {
    if (root == NULL) {
        return -1; 
    }
    int leftHeight = getHeight(root->left);
    int rightHeight = getHeight(root->right);
    return (leftHeight > rightHeight ? leftHeight : rightHeight) + 1;
}

/**
 * Helper function to recursively populate the string matrix.
 */
void fillMatrix(struct TreeNode* root, char*** res, int row, int col, int height) {
    if (root == NULL) {
        return;
    }
    
    // Allocate space and format the current node's value as a string
    res[row][col] = (char*)malloc(12 * sizeof(char)); 
    sprintf(res[row][col], "%d", root->val);
    
    // Only calculate offset and traverse if we are not at the leaf level
    if (row < height) {
        int offset = 1 << (height - row - 1);
        
        if (root->left) {
            fillMatrix(root->left, res, row + 1, col - offset, height);
        }
        if (root->right) {
            fillMatrix(root->right, res, row + 1, col + offset, height);
        }
    }
}

/**
 * Main solution function required by LeetCode.
 */
char*** printTree(struct TreeNode* root, int* returnSize, int** returnColumnSizes) {
    int height = getHeight(root);
    int rows = height + 1;
    int cols = (1 << (height + 1)) - 1; 
    
    *returnSize = rows;
    *returnColumnSizes = (int*)malloc(rows * sizeof(int));
    
    char*** res = (char***)malloc(rows * sizeof(char**));
    
    for (int i = 0; i < rows; i++) {
        (*returnColumnSizes)[i] = cols;
        res[i] = (char**)malloc(cols * sizeof(char*));
        
        for (int j = 0; j < cols; j++) {
            res[i][j] = (char*)malloc(1 * sizeof(char));
            res[i][j][0] = '\0'; // Properly terminate the empty string
        }
    }
    
    fillMatrix(root, res, 0, (cols - 1) / 2, height);
    
    return res;
}
