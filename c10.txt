#include <stdio.h>

int main() {
    // Array to store the frequency of digits 0 through 9
    int frequency[10] = {0};
    int ch;

    // Read character by character until the end of line or file
    while ((ch = getchar()) != '\n' && ch != EOF) {
        // Check if the character is a numeric digit
        if (ch >= '0' && ch <= '9') {
            // Map the character to its corresponding index (0-9)
            frequency[ch - '0']++;
        }
    }

    // Print the frequencies separated by a single space
    for (int i = 0; i < 10; i++) {
        printf("%d", frequency[i]);
        if (i < 9) {
            printf(" ");
        }
    }
    printf("\n");

    return 0;
}
