int numTrees(int n) {
    // dp[i] will store the number of unique BSTs that can be formed with i nodes
    int dp[n + 1];
    
    // Base cases
    dp[0] = 1; // An empty tree is 1 unique structural shape
    dp[1] = 1; // A tree with 1 node is 1 unique structural shape
    
    // Fill the DP table up to n
    for (int i = 2; i <= n; i++) {
        dp[i] = 0;
        // Consider each node 'j' as the root of the tree
        for (int j = 1; j <= i; j++) {
            // Left subtree will have (j - 1) nodes
            // Right subtree will have (i - j) nodes
            dp[i] += dp[j - 1] * dp[i - j];
        }
    }
    
    return dp[n];
}
