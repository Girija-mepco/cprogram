#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

// Define a structure for the Stack
typedef struct {
    int *data;
    int top;
    int capacity;
} Stack;

// Helper function to create a stack
Stack* createStack(int capacity) {
    Stack* stack = (Stack*)malloc(sizeof(Stack));
    stack->data = (int*)malloc(capacity * sizeof(int));
    stack->top = -1;
    stack->capacity = capacity;
    return stack;
}

// Helper function to push to a stack
void stackPush(Stack* stack, int x) {
    stack->data[++(stack->top)] = x;
}

// Helper function to pop from a stack
int stackPop(Stack* stack) {
    return stack->data[(stack->top)--];
}

// Helper function to peek the top of a stack
int stackPeek(Stack* stack) {
    return stack->data[stack->top];
}

// Helper function to check if a stack is empty
bool stackIsEmpty(Stack* stack) {
    return stack->top == -1;
}

// Define the MyQueue structure using two stacks
typedef struct {
    Stack* s1; // Used for push operations
    Stack* s2; // Used for pop/peek operations
} MyQueue;

// Initialize the queue
MyQueue* myQueueCreate() {
    MyQueue* queue = (MyQueue*)malloc(sizeof(MyQueue));
    int capacity = 10000; // Safe capacity for LeetCode constraints
    queue->s1 = createStack(capacity);
    queue->s2 = createStack(capacity);
    return queue;
}

// Push element x to the back of the queue
void myQueuePush(MyQueue* obj, int x) {
    stackPush(obj->s1, x);
}

// Removes the element from the front of the queue and returns it
int myQueuePop(MyQueue* obj) {
    if (stackIsEmpty(obj->s2)) {
        while (!stackIsEmpty(obj->s1)) {
            stackPush(obj->s2, stackPop(obj->s1));
        }
    }
    return stackPop(obj->s2);
}

// Returns the element at the front of the queue
int myQueuePeek(MyQueue* obj) {
    if (stackIsEmpty(obj->s2)) {
        while (!stackIsEmpty(obj->s1)) {
            stackPush(obj->s2, stackPop(obj->s1));
        }
    }
    return stackPeek(obj->s2);
}

// Returns true if the queue is empty, false otherwise
bool myQueueEmpty(MyQueue* obj) {
    return stackIsEmpty(obj->s1) && stackIsEmpty(obj->s2);
}

// Free the memory allocated for the queue
void myQueueFree(MyQueue* obj) {
    free(obj->s1->data);
    free(obj->s1);
    free(obj->s2->data);
    free(obj->s2);
    free(obj);
}
