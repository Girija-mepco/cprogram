/**
 * Note: The returned array must be malloced, assume caller calls free().
 */
int findBound(int* nums, int numsSize, int target, int isFirst) {
    int left = 0;
    int right = numsSize - 1;
    int bound = -1;
    
    while (left <= right) {
        int mid = left + (right - left) / 2;
        
        if (nums[mid] == target) {
            bound = mid; // Record the match found so far
            if (isFirst) {
                right = mid - 1; // Keep searching left for the starting position
            } else {
                left = mid + 1;  // Keep searching right for the ending position
            }
        } else if (nums[mid] < target) {
            left = mid + 1;
        } else {
            right = mid - 1;
        }
    }
    
    return bound;
}

int* searchRange(int* nums, int numsSize, int target, int* returnSize) {
    // LeetCode framework requires setting the return size
    *returnSize = 2;
    int* result = (int*)malloc(2 * sizeof(int));
    
    // Find the starting position and the ending position using binary search
    result[0] = findBound(nums, numsSize, target, 1);
    result[1] = findBound(nums, numsSize, target, 0);
    
    return result;
}
